// 选择排序
Array arr(0, n)		// 待排序的数组
for i = 0 to n-2	// 进行n-1趟
	k = i			
	// 从i的位置往后查找当前最大/小的数据
	for j = i+1 to n-1
		// 保存符合条件的数据的位置
		if arr[j] < arr[k]
			k = j
	// 更新第i大/小的数据
	if k != i
		arr[k] <==> arr[i]
end

		if ( temp < arr[j-1] )
